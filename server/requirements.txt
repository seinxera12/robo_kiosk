# Server runtime requirements for the voice kiosk chatbot.
#
# STT note: the LOCAL faster-whisper backend (STT_BACKEND=local) is NOT installed
# here — it pulls PyAV (`av`), which needs pkg-config + FFmpeg dev headers to
# build from source and bloats the image. The deployment uses the REMOTE STT
# backend (STT_BACKEND=remote), which only needs httpx + soundfile below.
# To run the local backend, also install requirements-local-stt.txt.

# Web framework
fastapi==0.104.1
uvicorn[standard]==0.24.0
websockets==12.0

# AI/ML models
# torch is intentionally NOT pinned here — it is pulled transitively by `kokoro`
# with a version compatible with that package. Pinning it separately risks a
# resolver conflict.
sentence-transformers==2.2.2
openai==1.3.0

# Vector database
chromadb==0.4.18

# HTTP client (also used by the remote STT backend)
httpx==0.25.1

# Audio processing (numpy + WAV encoding for the remote STT backend)
numpy==1.24.3
soundfile==0.12.1

# TTS engines
# Kokoro-82M: local in-process TTS (English primary, Japanese secondary).
# KokoClone (Japanese primary) runs as a separate microservice — not a pip dep.
kokoro>=0.9.4
misaki[en]>=0.9.4   # English G2P / phonemiser required by Kokoro

# Utilities
python-dotenv==1.0.0
pydantic==2.5.0
